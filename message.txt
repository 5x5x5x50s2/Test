-- DRAG FRAME MECHANICS
local function makeDraggable(frame, handle)
    local dragging = false
    local dragInput, dragStart, startPos

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            TweenService:Create(frame, TweenInfo.new(0.08, Enum.EasingStyle.Linear), {
                Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            }):Play()
        end
    end)
end

-- UI SHADOW EFFECT FUNCTION (Applies clean drop-shadow outlines)
local function applyUIShadow(parentInstance, thickness, transparency)
    local shadow = Instance.new("UIStroke")
    shadow.Color = Color3.fromRGB(0, 0, 0)
    shadow.Thickness = thickness or 2
    shadow.Transparency = transparency or 0.5
    shadow.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    shadow.Parent = parentInstance
end

local function createCyanButton(text, pos, parent)

    btn.Size = UDim2.new(1, 0, 0, 36)
    btn.Position = pos
    btn.BackgroundColor3 = Color3.fromRGB(0, 222, 255) -- Vibrant light-cyan match
    btn.BorderSizePixel = 0
    btn.Text = text
    btn.Font = Enum.Font.SourceSans
    btn.TextSize = 20
    btn.TextColor3 = Color3.fromRGB(0, 0, 0)
    btn.Parent = parent

    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 6)
    btnCorner.Parent = btn
    
    -- Apply custom button shadow
    applyUIShadow(btn, 2, 0.5)
    
    return btn
end

-- Text input clear feature

end)

-- Script string execution handling
executeBtn.MouseButton1Click:Connect(function()
    local targetFunc = loadstring or syn and syn.loadstring
    if targetFunc then
        local compiled, err = targetFunc(codeTextBox.Text)
        if compiled then
            task.spawn(compiled)
        else
            warn("Compilation Error: " .. tostring(err))
        end
    else
        warn("Execution environment runtime error: 'loadstring' unavailable.")
    end
end)

    end
end)
